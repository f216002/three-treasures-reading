rules_version = '2';

service firebase.storage {

    // v2 Sentence Bank teacher recordings (chinese-sentence-bank-v2).
    // Reads are public, same as the old Apps Script audio endpoint.
    // Writes need any Firebase Authentication sign-in; the v2 site uses
    // anonymous sign-in and the teacher PIN gates the recording UI.
    match /v2_audio/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // V4 教師錄音（舊路徑 v4_audio/xxx.webm）：公開讀取；寫入只限管理員。
    // 注意：Storage rules 無法查詢 Firestore 的核准名單，
    // 所以這裡只擋「未登入／匿名」，真正的審核把關在 Firestore rules
    //（錄音檔的路徑要寫進文件，那一步只有本人或管理員能做）。
    match /v4_audio/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
        && request.auth.token.email.lower() == 'f216002@gmail.com';
    }

    // V4 老師個人錄音（v4_audio/{uid}/xxx.webm）：公開讀取；
    // 寫入限本人（須 Google 登入＋信箱已驗證）或管理員。
    match /v4_audio/{teacherId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
        && request.auth.token.email_verified == true
        && (request.auth.uid == teacherId
          || request.auth.token.email.lower() == 'f216002@gmail.com');
    }
